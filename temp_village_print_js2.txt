                <script>
                function printVillageGroup(groupId) {
                    let htmlContent = "";
                    let title = "";
                    
                    if (groupId === "all") {
                        title = "All Church Villages";
                        ["Akoritho", "Philadelphia", "Bethsaida"].forEach(v => {
                            let tbl = document.getElementById("print-table-village-" + v);
                            if (tbl) htmlContent += "<h3 style='margin-top:30px;'>" + v + " Village</h3>" + tbl.outerHTML;
                        });
                    } else if (["Akoritho", "Philadelphia", "Bethsaida"].includes(groupId)) {
                        title = groupId + " Village Members";
                        let tbl = document.getElementById("print-table-village-" + groupId);
                        if (tbl) htmlContent += tbl.outerHTML;
                    } else if (groupId === "cleaners") {
                        title = "Church Cleaners - All Villages";
                        let tbl = document.getElementById("print-table-cleaners");
                        if (tbl) htmlContent += tbl.outerHTML;
                    } else if (groupId === "cookers") {
                        title = "Church Cookers - All Villages";
                        let tbl = document.getElementById("print-table-cookers");
                        if (tbl) htmlContent += tbl.outerHTML;
                    }

                    if (!htmlContent) {
                        alert("No data available to print for this selection. (Tables not found)");
                        return;
                    }

                    var w = window.open("", "_blank");
                    if (!w) { alert("Popup blocked! Please allow popups for this page."); return; }
                    
                    let curDate = new Date().toLocaleString();
                    
                    w.document.write("<html><head><title>Print " + title + "</title>");
                    w.document.write("<style>");
                    w.document.write("body { font-family: Arial, sans-serif; color: #333; margin: 20px; }");
                    w.document.write("table { width: 100%; border-collapse: collapse; margin-top: 15px; font-size: 0.85rem; }");
                    w.document.write("th, td { border: 1px solid #ccc; padding: 8px; text-align: left; }");
                    w.document.write("th { background: #f3f4f6; color: #333; font-weight: bold; }");
                    w.document.write("img { width: 40px; height: 40px; border-radius: 50%; object-fit: cover; }");
                    w.document.write(".badge { display: inline-block; padding: 2px 6px; border-radius: 4px; font-size: 0.75rem; background: #eee; }");
                    w.document.write("h3 { color: #444; border-bottom: 2px solid #ccc; padding-bottom: 5px; }");
                    w.document.write("* { -webkit-print-color-adjust: exact !important; print-color-adjust: exact !important; color-adjust: exact !important; }");
                    w.document.write("@media print { @page { margin: 15mm; } body { margin: 0; } }");
                    w.document.write(".watermark { position:fixed; top:50%; left:50%; transform:translate(-50%,-50%) rotate(-45deg); font-size:5rem; color:rgba(0,0,0,0.08); font-weight:bold; white-space:nowrap; z-index:-1; pointer-events:none; }");
                    w.document.write("</style></head><body>");
                    
                    w.document.write("<div class='watermark'>E.A.P.C MUNYARI CHURCH</div>");
                    
                    if (groupId === "all") {
                        let headerHtml = document.getElementById("print-village-headers").innerHTML;
                        w.document.write(headerHtml);
                        w.document.write("<h2 style='text-align:center; margin:30px 0 10px 0; color:#444;'>" + title + "</h2>");
                        w.document.write("<p style='text-align:center; margin:0 0 20px 0; font-size:0.8rem; color:#777;'>Generated on: " + curDate + "</p>");
                    } else {
                        w.document.write("<div style='text-align:center; margin-bottom: 30px;'>");
                        w.document.write("<img src='church_logo.jpg' style='width:80px; height:80px; border-radius:50%; margin-bottom:10px;'>");
                        w.document.write("<h2 style='margin:0; color:#1e3a8a;'>E.A.P.C MUNYARI CHURCH</h2>");
                        w.document.write("<h3 style='margin:5px 0 0 0; border:none; color:#555;'>" + title + "</h3>");
                        w.document.write("<p style='margin:5px 0 0 0; font-size:0.8rem; color:#777;'>Generated on: " + curDate + "</p>");
                    }
                    
                    w.document.write(htmlContent);
                    
                    if (groupId === "all") {
                        let footerHtml = document.getElementById("print-village-signatures").innerHTML;
                        w.document.write(footerHtml);
                    } else {
                        w.document.write("<div style='margin-top:50px; display:flex; justify-content:space-between; font-size:0.9rem;'>");
                        w.document.write("<div><p>Prepared by:</p><p style='margin-bottom:10px;'>_______________________</p><p style='font-size:0.8rem; color:#555;'>Sign / Date</p></div>");
                        w.document.write("<div><p>Approved by (Pastor):</p><p style='margin-bottom:10px;'>_______________________</p><p style='font-size:0.8rem; color:#555;'>Sign / Date</p></div>");
                        w.document.write("</div>");
                    }
                    
                    w.document.write("<div style='position:fixed; bottom:10px; left:0; right:0; text-align:center; font-size:0.75rem; color:#777; font-style:italic;'>Generated from Munyari Portal</div>");
                    w.document.write("</body></html>");
                    w.document.close();
                    w.focus();
                    setTimeout(function(){ w.print(); }, 800);
                    
                    document.getElementById('villagePrintModal').style.display='none';
                }
                </script>
